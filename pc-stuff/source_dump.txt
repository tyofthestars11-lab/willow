#include "CarryWeightSystem.h"

UCarryWeightSystem::UCarryWeightSystem()
{
    PrimaryComponentTick.bCanEverTick = false;
    bUnlimitedCarryWeight = true;
}

void UCarryWeightSystem::BeginPlay()
{
    Super::BeginPlay();
}

bool UCarryWeightSystem::AddItem(const FInventoryItem& NewItem)
{
    // Unlimited carry weight - always allow adding items
    for (FInventoryItem& Item : Items)
    {
        if (Item.Name == NewItem.Name && Item.Rarity == NewItem.Rarity)
        {
            Item.Quantity += NewItem.Quantity;
            return true;
        }
    }
    Items.Add(NewItem);
    return true;
}

bool UCarryWeightSystem::RemoveItem(const FString& ItemName, int32 Quantity)
{
    for (int32 i = Items.Num() - 1; i >= 0; i--)
    {
        if (Items[i].Name == ItemName)
        {
            Items[i].Quantity -= Quantity;
            if (Items[i].Quantity <= 0)
            {
                Items.RemoveAt(i);
            }
            return true;
        }
    }
    return false;
}

float UCarryWeightSystem::GetCurrentWeight() const
{
    float TotalWeight = 0.0f;
    for (const FInventoryItem& Item : Items)
    {
        TotalWeight += Item.Weight * Item.Quantity;
    }
    return TotalWeight;
}

TArray<FInventoryItem> UCarryWeightSystem::GetItems() const
{
    return Items;
}

bool UCarryWeightSystem::HasItem(const FString& ItemName) const
{
    for (const FInventoryItem& Item : Items)
    {
        if (Item.Name == ItemName)
        {
            return true;
        }
    }
    return false;
}

int32 UCarryWeightSystem::GetItemCount() const
{
    return Items.Num();
}

UGodModeAssetManager::UGodModeAssetManager()
{
    PrimaryComponentTick.bCanEverTick = false;
    CurrentGlowMaterial = nullptr;
    CurrentAuraMaterial = nullptr;
    LoadedTexture = nullptr;
    DefaultGlowColor = FLinearColor(0.0f, 1.0f, 0.5f, 1.0f);
    DefaultGlowIntensity = 5.0f;
    DefaultAuraColor = FLinearColor(0.5f, 0.0f, 1.0f, 1.0f);
    DefaultAuraOpacity = 0.7f;
    DefaultPulseSpeed = 2.0f;
}

void UGodModeAssetManager::BeginPlay()
{
    Super::BeginPlay();
    SetupDefaultMaterials();
}

UMaterialInstanceDynamic* UGodModeAssetManager::CreateGlowMaterial(UMaterialInterface* BaseMaterial, FLinearColor GlowColor, float GlowIntensity)
{
    if (!BaseMaterial)
    {
        return nullptr;
    }

    UMaterialInstanceDynamic* DynMaterial = UMaterialInstanceDynamic::Create(BaseMaterial, this);
    if (DynMaterial)
    {
        DynMaterial->SetVectorParameterValue(FName("GlowColor"), GlowColor);
        DynMaterial->SetScalarParameterValue(FName("GlowIntensity"), GlowIntensity);
        CurrentGlowMaterial = DynMaterial;
    }
    return DynMaterial;
}

UMaterialInstanceDynamic* UGodModeAssetManager::CreateAuraMaterial(UMaterialInterface* BaseMaterial, FLinearColor AuraColor, float Opacity, float PulseSpeed)
{
    if (!BaseMaterial)
    {
        return nullptr;
    }

    UMaterialInstanceDynamic* DynMaterial = UMaterialInstanceDynamic::Create(BaseMaterial, this);
    if (DynMaterial)
    {
        DynMaterial->SetVectorParameterValue(FName("AuraColor"), AuraColor);
        DynMaterial->SetScalarParameterValue(FName("Opacity"), Opacity);
        DynMaterial->SetScalarParameterValue(FName("PulseSpeed"), PulseSpeed);
        CurrentAuraMaterial = DynMaterial;
    }
    return DynMaterial;
}

UTexture2D* UGodModeAssetManager::LoadTexture(const FString& TexturePath)
{
    UTexture2D* Texture = Cast<UTexture2D>(StaticLoadObject(UTexture2D::StaticClass(), nullptr, *TexturePath));
    if (Texture)
    {
        LoadedTexture = Texture;
    }
    return Texture;
}

void UGodModeAssetManager::SetupDefaultMaterials()
{
    // Default materials will be set up when a valid base material is provided
    CurrentGlowMaterial = nullptr;
    CurrentAuraMaterial = nullptr;
}

UMaterialInstanceDynamic* UGodModeAssetManager::GetGlowMaterial() const
{
    return CurrentGlowMaterial;
}

UMaterialInstanceDynamic* UGodModeAssetManager::GetAuraMaterial() const
{
    return CurrentAuraMaterial;
}

{
    PrimaryComponentTick.bCanEverTick = false;
    DropRate = 1.0f;
    MaxDrops = 99;
    bGuaranteedDrop = true;
}

void UGodModeLootSystem::BeginPlay()
{
    Super::BeginPlay();
}

FString UGodModeLootSystem::GenerateLootDrop()
{
    if (LootTable.Num() == 0)
    {
        return FString(TEXT("None"));
    }
    if (bGuaranteedDrop)
    {
        int32 Index = FMath::RandRange(0, LootTable.Num() - 1);
        return LootTable[Index];
    }
    float Roll = FMath::FRand();
    if (Roll <= DropRate)
    {
        int32 Index = FMath::RandRange(0, LootTable.Num() - 1);
        return LootTable[Index];
    }
    return FString(TEXT("None"));
}

void UGodModeLootSystem::AddToLootTable(const FString& ItemName)
{
    LootTable.Add(ItemName);
}

void UGodModeLootSystem::ClearLootTable()
{
    LootTable.Empty();
}

#include "SimplePuzzleSystem.h"

USimplePuzzleSystem::USimplePuzzleSystem()
{
    PrimaryComponentTick.bCanEverTick = false;
    bIsSolved = false;
    PuzzleName = TEXT("DefaultPuzzle");
    RequiredSteps = 3;
    CurrentStep = 0;
}

void USimplePuzzleSystem::BeginPlay()
{
    Super::BeginPlay();
}

void USimplePuzzleSystem::AdvanceStep()
{
    if (!bIsSolved)
    {
        CurrentStep++;
        CheckSolved();
    }
}

void USimplePuzzleSystem::ResetPuzzle()
{
    CurrentStep = 0;
    bIsSolved = false;
}

bool USimplePuzzleSystem::CheckSolved()
{
    if (CurrentStep >= RequiredSteps)
    {
        bIsSolved = true;
    }
    return bIsSolved;
}
